$dir = "\\阿林\协作\仓库\lsy"
$git = "C:\Program Files\Git\bin\git.exe"

Get-ChildItem $dir -Filter "*.html" | ForEach-Object {
    $name = $_.Name
    $batName = "同步$name.bat"
    
    $content = @"
@echo off
chcp 65001 >nul
cd /d "$dir"
"$git" add "$name"
"$git" commit -m "更新 $name"
"$git" push
echo $name 同步完成！
pause
"@
    
    Set-Content -Path "$dir\$batName" -Value $content -Encoding UTF8
    Write-Host "已生成: $batName"
}
